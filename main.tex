\begin{itemize}
    \item Gain en tension :
    $$A_v = \frac{v_O}{v_I} \quad \left[\unit{\volt\per\volt}\right]$$
    \item Gain en courant :
    $$A_i = \frac{i_O}{i_I} \quad \left[\unit{\ampere\per\ampere}\right]$$

    $$A_p = A_v\cdot A_i \quad \left[\unit{\watt\per\watt}\right]$$

    $$\eta = \frac{P_L}{P_{dc}} \times 100 \quad \left[\%\right]$$
    
    $P_L$ : Puissance sur la charge

    $P_{dc} = V_{CC}\cdot I_{CC} + V_{EE}\cdot I_{EE}$
\end{itemize}

$$ i = I_S \cdot \left(e^{v / (n \cdot V_T)} - 1\right) \quad \text{avec} \quad V_T = \frac{k \cdot T}{q} $$

$n$ : Constante empirique entre $1$ et $2$

$k$ : Constante de Boltzmann ($1.38 \; 10^{-23}$)

$T$ : Température en kelvins

$q$ : Charge électronique élémentaire ($1.6 \; 10^{-19}$)

$$I_D = \frac{V_DD - V_D}{R}$$

\subsubsection{Modèle petit-signal}

Cas de superposition d'un signal de polarisation (dc) et d'un signal alternatif (ac)

$$v_D(t) = V_D + v_d(t)$$
$$i_D(t) = I_D + i_d(t)$$

Concrètement, pour la partie AC, on peut remplacer la diode par une résistance $r_d$
$$r_d = \frac{1}{\left.\frac{\partial i_D}{\partial v_D}\right|_{i_D = I_D}} = \frac{(n) V_T}{I_D}$$

(Fonctionne bien jusqu'à des tensions alternatives d'environ $100 \unit{m\volt}$)

\medskip

En petit-signal, les sources DC sont des masses

\subsection{Transistor NMOS}

\begin{center} \includegraphics[scale=0.35]{images/symbol-nmos.png} \end{center}

\begin{center} \includegraphics[scale=0.4]{images/mode-nmos.png} \end{center}

%\begin{center} \includegraphics[scale=0.45]{images/nmos-eq.png} \end{center}

$$\boxed{v_{OV}  = v_{GS} - V_{tn}}$$

\begin{itemize}
    \item \textbf{Cut-off} (coupé) : $ \quad v_{GS} < V_{tn}$
    \item \textbf{Overdrive} (passant) : $ \quad v_{GS} \geq V_{tn}$
    
    \begin{itemize}
        \item \textbf{Saturation} : $ \quad v_{DS} \geq v_{OV} \Leftrightarrow v_{GD} \leq V_{tn}$
        
        $$\boxed{i_D = \frac{1}{2} k_n' \frac{W}{L}v_{OV}^2 }\cdot \underbrace{(1 + \lambda_n\cdot v_{DS})}_\text{Effet d'Early}$$

        \item \textbf{Triode} : $ \quad v_{DS} < v_{OV} \Leftrightarrow v_{GD} > V_{tn}$
        
        $$\boxed{i_D = k_n' \frac{W}{L}\left[v_{OV} - \frac{1}{2} v_{DS}\right] \cdot v_{DS}}$$
    \end{itemize}
\end{itemize}

($k_n' = \mu_n C_{ox}$, $W$ et $L$ sont des paramètres du NMOS)

$k_n \text{ ou } \beta = k_n' \cdot W / L$

À $v_{DS}$ faible et $v_{GS}$ élevé, $i_D$ quasi-linéaire $\rightarrow$ résistance MOS

    \end{itemize}
\end{itemize}

\subsubsection{En 'grand signal'}

\begin{center} \includegraphics[scale=0.45]{images/ampli1.png} \end{center}

$$v_O = v_{DS} = V_{DD} - R_D\cdot i_D$$

\begin{itemize}
    \item $v_{GS} \leq V_t$ : $i_D = 0 \quad ; \quad v_O = V_{DD}$
    \item $V_t < V_{GS} \leq V_{GS|_B}$ càd $v_O > v_{GS} - V_t$ :
    $$v_O = V_{DD} - R_D \cdot \frac{1}{2} k_n(v_{GS} - V_t)^2 \qquad \text{(saturation, sans Early)}$$
    $$\Rightarrow V_{GS|_B} = V_t + \frac{\sqrt{2 k_n R_D V_{DD} + 1} - 1}{k_n R_D} \qquad \text{pour} \quad v_{DS} = v_{OV}$$
    \item $v_{GS} > V_{GS|_B}$ : Triode
    $$v_O = V_{DD} - i_D(v_{GS},v_O)$$
\end{itemize}

\subsubsection{En 'petit-signal'}

$$v_{GS} = V_{GS} + v_{gs}$$
\begin{flalign*}
    \Rightarrow i_D &= \frac{1}{2} k_n (V_{GS} + v_{gs} - V_t)^2\\
    &= \underbrace{\frac{1}{2}k_n (V_{GS} - V_t)^2}_{I_D \quad \text{(DC)}} + \underbrace{k_n (V_{GS} - V_t) v_{gs}}_{i_d \quad \text{(ac)}} + \underbrace{\frac{1}{2} k_n v_{gs}^2}_\text{distortion non-linéaire}
\end{flalign*}
On va faire en sorte que la partie non-linéaire soit très petite

\medskip

Pour $v_{gs} << 2\cdot V_{OV}$ :

$$i_d = k_n(V_{GS} - V_t) v_{gs} = g_m \cdot v_{gs}$$
$$g_m \equiv \frac{i_d}{v_{gs}} = k_n (V_{GS} - V_t) = k_n V_{OV}$$

$g_m$ : Transconductance petit-signal

$$g_m \equiv \left.\frac{\partial i_D}{\partial v_{GS}}\right|_{v_{GS} = V_{GS}}$$

On peut aussi trouver que
$$g_m = \frac{2\cdot I_D}{V_{OV}}$$

\medskip

\textbf{Gain petit-signal} :

$$v_{ds} = -i_d\cdot R_D = -g_m \cdot v_{gs} \cdot R_D$$

$$A_v \equiv \frac{v_{ds}}{v_{gs}} = -g_m \cdot R_D$$

On peut passer en petit-signal à une représentation avec une matrice de quadripôle de transconductance

\textbf{Effet Early} :

Il faut rajouter l'effet Early, car il n'est pas négligable en AC

\begin{align*}
g_m &\equiv \frac{\partial i_D}{\partial v_{GS}}\Bigg|_{V_{GS}, V_{DS}} = \frac{2I_{DS}}{V_{OV}} \\
g_d &\equiv \frac{\partial i_D}{\partial v_{DS}}\Bigg|_{V_{GS}, V_{DS}} = \lambda I_{DS}
\end{align*}

$$r_O = g_d^{-1} = \left(\frac{1}{2} k_n (V_{GS} - V_t)^2 \cdot \lambda\right)^{-1} = \frac{1}{\lambda \cdot I_D} = \frac{V_A}{I_D}$$

\begin{center} \includegraphics[scale=0.45]{images/ampli1-early.png} \end{center}

$$A_v \equiv \frac{v_{ds}}{v_{gs}} = -g_m \cdot (R_D || r_0)$$
